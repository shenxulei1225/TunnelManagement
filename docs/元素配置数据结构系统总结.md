### 2. 属性定义系统

每个控件属性都包含完整的元数据：

```typescript
interface PropertyDefinition {
  name: string                    // 属性名
  type: DataType                  // 数据类型
  required?: boolean              // 是否必需
  defaultValue?: any              // 默认值
  description: string             // 描述
  
  // 验证规则
  validation?: {
    min?: number                  // 最小值/长度
    max?: number                  // 最大值/长度
    pattern?: RegExp              // 正则表达式
    enum?: any[]                  // 枚举值
    custom?: (value: any) => boolean | string  // 自定义验证
  }
  
  // UI配置
  ui?: {
    label: string                 // 显示标签
    placeholder?: string          // 占位符
    helpText?: string            // 帮助文本
    group?: string               // 分组
    order?: number               // 排序
    controlType?: string         // 控件类型
    options?: Array<{label: string, value: any}>  // 选项列表
  }
  
  // 主题映射
  theme?: {
    cssVariable?: string          // CSS变量名
    cssProperty?: string          // CSS属性名
    tokenName?: string           // Design Token名称
    category?: 'color' | 'size' | 'spacing' | 'typography'
  }
  
  // 业务映射
  business?: {
    fieldType?: 'identifier' | 'display' | 'input' | 'computed'
    dataSource?: string          // 数据源
    formatter?: string           // 格式化器
    validator?: string           // 验证器
  }
}
```

## 🎯 核心功能

### 1. 动态表单生成

基于组件定义自动生成配置表单：

```vue
<!-- 自动按属性分组 -->
<template v-for="group in propertyGroups" :key="group.name">
  <el-collapse-item :title="group.name">
    <template v-for="property in group.properties">
      <!-- 根据属性类型自动选择控件 -->
      <el-switch v-if="property.type === 'boolean'" />
      <el-select v-else-if="property.type === 'enum'" />
      <el-input-number v-else-if="property.type === 'number'" />
      <el-input v-else />
    </template>
  </el-collapse-item>
</template>
```

### 2. 实时验证

```typescript
const handlePropertyChange = (propertyName: string, value: any) => {
  componentValues.value[propertyName] = value
  
  // 实时验证
  const property = currentComponent.value.props.find(p => p.name === propertyName)
  if (property) {
    const result = Validator.validateProperty(value, property)
    if (!result.valid) {
      showValidation.value = true
    }
  }
}
```

### 3. 智能代码生成

```typescript
const generateCode = () => {
  const code = CodeGenerator.generateVueCode(
    currentComponent.value,
    componentValues.value,
    {
      options: {
        typescript: true,
        composition: true,
        script: true,
        style: false,
        imports: true
      }
    }
  )
  generatedCode.value = code
}
```
