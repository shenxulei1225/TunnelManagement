### 4. 路由生成逻辑增强

#### 4.1 动态路由检测
```typescript
// 解析动态路由配置
const dynamicConfig = route.dynamicConfig
const isDynamicRoute = dynamicConfig && 
  dynamicConfig.parameterName && 
  route.path.includes(':')

// 设置路由元数据
const meta = {
  // ... 原有字段
  isDynamicRoute,
  paramName: dynamicConfig?.parameterName,
  titleTemplate: dynamicConfig?.titleTemplate,
  multiInstance: dynamicConfig?.enableMultiInstance,
  cacheStrategy: isDynamicRoute ? 'param-based' : 'default'
}
```

### 5. TagsView 动态标题支持

#### 5.1 参数化标题处理
```typescript
// 处理动态路由标题
if (view.meta?.isDynamicRoute && view.meta?.titleTemplate && view.params) {
  const template = view.meta.titleTemplate as string
  const paramName = view.meta.paramName as string
  const paramValue = view.params[paramName]
  
  if (paramValue) {
    // 替换模板中的参数占位符
    title = template
      .replace(new RegExp(`\\$\\{${paramName}\\}`, 'g'), paramValue as string)
      .replace(new RegExp(`:${paramName}`, 'g'), paramValue as string)
  }
}
```

const dynamicMenuConfig = {

  parentId: 1,
  
  // 路径配置 - 关键改进
  path: "dynamic-classification/:configId",  // ✅ 支持动态参数

  componentName: "DragDropClassificationDynamic",

  keepAlive: true,
  alwaysShow: false
}

## 🔧 使用指南

### 1. 创建动态菜单

```javascript
// 使用 DynamicMenuHelper 创建
const config = DynamicMenuHelper.createDynamicClassificationMenu({
  name: "设备分类管理",
  paramName: "deviceCategoryId", 
  titleTemplate: "设备分类 - ${deviceCategoryId}",
  parentId: 1,
  sort: 100
})

// 转换为后端API格式
const backendFormat = DynamicMenuHelper.toBackendFormat(config)

// 调用后端API创建菜单
await createMenu(backendFormat)
```

### 2. 访问动态路由

```javascript
// 打开不同的实例
router.push('/dynamic-classification/device-category')    // 设备分类
router.push('/dynamic-classification/product-category')  // 产品分类
router.push('/dynamic-classification/content-category')  // 内容分类

// 每个实例将显示不同的标题
// "动态分类 - device-category"
// "动态分类 - product-category" 
// "动态分类 - content-category"
```
